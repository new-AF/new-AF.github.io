import { readdir } from "node:fs/promises";

export const listDir = async (path) => {
    try {
        const array = await readdir(path);
        return { success: true, array };
    } catch (error) {
        return {
            success: false,
            array: [],
            error,
        };
    }
};
jsxundefined Search Bar (AI assisted)
import Fuse from "fuse.js";
import {
    Combobox,
    ComboboxButton,
    ComboboxInput,
    ComboboxOption,
    ComboboxOptions,
} from "@headlessui/react";
import { CheckIcon, ChevronDownIcon } from "@heroicons/react/20/solid";
import { MagnifyingGlassIcon } from "@heroicons/react/24/outline";

import { useState } from "react";
import { mergeClassNames } from "simple-merge-class-names";
import { useEffect } from "react";

const people = [
    { id: 1, name: "Tom Cook", tags: ["javascript", "typescript"] },
    { id: 2, name: "Wade Cooper", tags: ["react"] },
    { id: 3, name: "Tanya Fox", tags: ["php"] },
    { id: 4, name: "Arlene Mccoy", tags: ["html", "css"] },
    { id: 5, name: "Devon Webb", tags: ["c++"] },
];

export default function Example({ setResults }) {
    const [query, setQuery] = useState("");
    const [loading, setLoading] = useState(false);

    const search = () => {
        const fuse = new Fuse(people, { keys: ["name", "tags"] });
        const results = fuse.search(query);
        setResults(results);
        console.log(results);
    };

    useEffect(() => {
        if (query === "") {
            setResults([]);
            setLoading(false);
            return;
        }

        // debounce
        const id = setTimeout(() => {
            setLoading(false);
            search();
        }, 300);

        setLoading(true);

        return () => clearTimeout(id);
    }, [query]);

    return (
        <div className="mx-auto w-full">
            <Combobox
                value={query}
                // onClose={() => setQuery("")}
            >
                <div className="group relative">
                    <ComboboxInput
                        className={mergeClassNames(
                            "w-full rounded-lg border-none bg-white/5 py-4 pr-8 pl-8 text-md/6 text-(--custom-color-text)",
                            "focus:not-data-focus:outline-none data-focus:outline-2 data-focus:-outline-offset-2 data-focus:outline-white/25",
                        )}
                        value={query}
                        onChange={(event) => setQuery(event.target.value)}
                    />
                    <ComboboxButton className=" absolute inset-y-0 right-0 px-6">
                        {loading ? (
                            <span className="daisy-loading daisy-loading-spinner daisy-loading-md"></span>
                        ) : (
                            <MagnifyingGlassIcon className="size-4 text-gray-400 group-focus-within:text-gray-300 transition-colors" />
                        )}
                    </ComboboxButton>
                </div>
            </Combobox>
        </div>
    );
}
