React + Headless UI

12import { useState, useEffect } from "react";
13import { mergeTailwindClassNames } from "../../../../utils";
14
15const people = [
16    { id: 1, name: "Tom Cook", tags: ["javascript", "typescript"] },
17    { id: 2, name: "Wade Cooper", tags: ["react"] },
18    { id: 3, name: "Tanya Fox", tags: ["php"] },
19    { id: 4, name: "Arlene Mccoy", tags: ["html", "css"] },
20    { id: 5, name: "Devon Webb", tags: ["c++"] },
21];
22
23export default function Example({ setResults }) {
24    const [query, setQuery] = useState("");
25    const [loading, setLoading] = useState(false);
26
27    const search = () => {
28        const fuse = new Fuse(people, { keys: ["name", "tags"] });
29        const results = fuse.search(query);
30        setResults(results);
31        console.log(results);
32    };
33
34    useEffect(() => {
35        if (query === "") {
36            setResults([]);
37            setLoading(false);
38            return;
39        }
40
41        // debounce
42        const id = setTimeout(() => {
43            setLoading(false);
44            search();
45        }, 300);
46
47        setLoading(true);
48
49        return () => clearTimeout(id);
50    }, [query]);
51
52    return (
53        <div className="mx-auto w-full">
54            <Combobox
55                value={query}
56                // onClose={() => setQuery("")}
57            >
58                <div className="group relative">
59                    <ComboboxInput
60                        className={mergeTailwindClassNames(
61                            "w-full rounded-lg border-none bg-white/5 py-4 pr-8 pl-8 text-md/6 text-(--custom-color-text)",
62                            "focus:not-data-focus:outline-none data-focus:outline-2 data-focus:-outline-offset-2 data-focus:outline-white/25",
63                        )}
64                        value={query}
65                        onChange={(event) => setQuery(event.target.value)}
66                    />
67                    <ComboboxButton className=" absolute inset-y-0 right-0 px-6">
68                        {loading ? (
69                            <span className="daisy-loading daisy-loading-spinner daisy-loading-md"></span>
70                        ) : (
71                            <MagnifyingGlassIcon className="size-4 text-gray-400 group-focus-within:text-gray-300 transition-colors" />
72                        )}
73                    </ComboboxButton>
74                </div>
75            </Combobox>
76        </div>
