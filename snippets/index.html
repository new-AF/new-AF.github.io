23export default function Example({ setResults }) {
24    const [query, setQuery] = useState("");
25    const [loading, setLoading] = useState(false);
26
27    const search = () => {
28        const fuse = new Fuse(people, { keys: ["name", "tags"] });
29        const results = fuse.search(query);
30        setResults(results);
31        console.log(results);
32    };
33
34    useEffect(() => {
35        if (query === "") {
36            setResults([]);
37            setLoading(false);

42        const id = setTimeout(() => {

47        setLoading(true);
48
