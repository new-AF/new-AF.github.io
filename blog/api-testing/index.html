import { PORT } from "@/config/environment";
import { userRouter } from "@/routes/user.route";
import { logRequest } from "./middleware/logRequest";
import { checkApiKey } from "./middleware/checkApiKey";
import { errorHandler } from "./middleware/errorHandler";

export const app = express();

// --- Middleware ----
// 1) Log every request
app.use(logRequest);

// 2) Convert request body to JSON
app.use(express.json());

// 3) Authorization
app.use(checkApiKey);

// 4) Controllers
app.use("/api/v1/users", userRouter);

// 5) Handle Errors
app.use(errorHandler);

app.listen(PORT, () => {
    console.log(
        `🚀 Day 7: API Testing server running on http://localhost:${PORT}`
    );
});

Better DX Testing Approach

A tedious and repetitive part of the current approach that doesn't scale is always coming up with unique and descriptive function names (e.g. ts: createUserValid) and having to again replicate the description in ts: it("creates a user with valid fields", createUserValid), example:

TypeScript
// create a user with valid fields
const createUserValid = async () => {
    const response = await request(app)
        .post("/api/v1/users")
        // ...

    // resource created
    expect(response.status).toBe(201);
    // ...

// top-level describe, repeat same description
describe("POST /api/v1/users", () => {
    it("creates a user with valid fields", createUserValid);
});

Instead we will pursue an approach with better developer experience (DX) that's use easy and equally descriptive, it uses object key names as description, and value as test function, example:

TypeScript
// all our test cases to be iterated through
const cases = {};

// create a user with valid fields
cases["creates valid user, returns 200"] = async () => {
    const response = await request(app).post("/api/v1/users");
    // ...

    // resource created
    expect(response.status).toBe(201);
    // ...
};

// top-level describe
describe("POST /api/v1/users", () =>
    Object.entries(cases).forEach(([caseName, testFunction]) =>
        it(caseName, testFunction)
    ));

The test case that failed is:

  • DELETE /api/v1/users/abc

  • It should return error status code 400 because it is a (Zod) runtime validation error, not 500 an internal server error.

  • This is because we only accept numbers as request parameters (DELETE /api/v1/users/1), not strings.

  • Our error middleware should be activated but is not happening:

    src/middleware/errorHandler.ts
    export const errorHandler = (error: unknown, response: Response, ) => {
    // Validation error
    if (error instanceof ZodError) {
        // Bad request
        response.status(400);
        response.json({ errors: flattenError(error) });
        return;
    }

    // Standard JS Exception
    if (error instanceof Error) {
        // Internal Server Error
        response.status(500);
        response.json({ error: error.message });
        return;
    }

    // ...

Turns out the bug is in the errorHandler function signature: ts: const errorHandler = (error: unknown, response: Response, ) => ...

This is because Express only treats 4-parameter functions as middleware, but we have 2 which means ours is never called.

This means: this whole time we thought our error-handling middleware was registered and would catch errors but it was never called, not even once.

So let's fix our errorHandler function signature to be 4-parameter:

src/middleware/errorHandler.ts
import { Request, Response, NextFunction } from "express";
import { ZodError, flattenError } from "zod";

export const errorHandler = (
    error: unknown,
    _request: Request,
    response: Response,
    _next: NextFunction
) => {
    // Validation error
    if (error instanceof ZodError) {
        // Bad request
        response.status(400);
        response.json({ errors: flattenError(error) });
        return;
    }

    // Standard JS Exception
    if (error instanceof Error) {
        // Internal Server Error
        response.status(500);
        response.json({ error: error.message });
        return;
    }

    /*
        Safety net because JavaScript lets us throw literally anything, we must guard against it.
        e.g. `throw "DB Crashed", throw 1234`
        Without this fallback, Express would crash or hang if something bizarre was thrown.
    */
    response.status(500);
    response.json({
        error: "Unknown server error",
        detail: typeof error === "string" ? error : undefined,
    });
};

And voila all our test cases pass!
