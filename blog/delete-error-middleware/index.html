Create Error Handling Middleware

Middleware is any function between the HTTP request and response, we've seen a few before in src/server.ts:

  • ts: app.use(express.json()); Converts request body byte-stream to JSON that we access by ts: request.body in controllers.

  • ts: app.use("/api/v1/users", userRouter); Our router which maps the users endpoint to relevant controllers.

We'll add another middleware and use it as the following: Upon error e.g. Validation Errors of invalid request payloads or query parameters we'll throw and pass the error from the controller to the error-handling middleware.

  • But first create src/middleware Folder

TypeScript
1/* fileName: src/middleware/errorHandler.ts */
2import { Request, Response, NextFunction } from "express";
3import { ZodError, flattenError } from "zod";
4
5export const errorHandler = (
6    error: unknown,
7    _request: Request,
8    response: Response,
9    _next: NextFunction
10) => {
11    // Validation error
12    if (error instanceof ZodError) {
13        // Bad request
14        response.status(400);
15        response.json({ errors: flattenError(error) });
16        return;
17    }
18
19    // Standard JS Exception
20    if (error instanceof Error) {
21        // Internal Server Error
22        response.status(500);
23        response.json({ error: error.message });
24        return;
25    }
26
27    /*
28        Safety net because JavaScript lets us throw literally anything, we must guard against it.
29        e.g. `throw "DB Crashed", throw 1234`
30        Without this fallback, Express would crash or hang if something bizarre was thrown.
31    */
32    response.status(500);
33    response.json({
34        error: "Unknown server error",
35        detail: typeof error === "string" ? error : undefined,
36    });
37};

Add DELETE Controller

TypeScript
1/* fileName: src/controllers/deleteUser.ts */
2import { NextFunction, Request, Response } from "express";
3import { users } from "@/data/users.store";
4import { userIdParamsSchema } from "@/validators/users/user.params";
5
6const deleteUserCore = (request: Request, response: Response) => {
7    const { params } = request;
8    const paramsResult = userIdParamsSchema.safeParse(params);
9
10    if (paramsResult.success === false) {
11        // Zod Error, handled by `errorHandler` middleware
12        throw paramsResult.error;
13    }
14
15    const userId = paramsResult.data.id;
16    const findIndex = users.findIndex((user) => userId === user.id);
17
18    // User not found. Not an error (Runtime or Validation), just expected outcome.
19    if (findIndex === -1) {
20        response.status(404);
21        response.json({
22            message: response.json({
23                message: `User with id ${userId} not found`,
24            }),
25        });
26        return;
27    }
28
29    // Delete users[findIndex]
30    users.splice(findIndex, 1);
31
32    // Success, no content
33    response.status(204);
34    response.send();
35};
36
37export const deleteUser = (
38    request: Request,
39    response: Response,
40    next: NextFunction
41) => {
42    try {
43        deleteUserCore(request, response);
44    } catch (error) {
45        // Call error handler middleware
46        next(error);
47    }
48};
