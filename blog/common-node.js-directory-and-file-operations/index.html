📖 Read File Line By Line

Unknown Code
1/* fileName: readLines.mjs */
2
3import { createReadStream } from "node:fs";
4import { createInterface } from "node:readline/promises";
5
6export const readLines = async function* (filePath, encoding = "utf8") {
7    let stream;
8    let lineReader;
9
10    try {
11        stream = createReadStream(filePath, encoding);
12        lineReader = createInterface({ input: stream, crlfDelay: Infinity });
13
14        for await (const line of lineReader) {
15            yield { success: true, line, error: null };
16        }
17    } catch (error) {
18        yield {
19            success: false,
20            line: null,
21            error,
22        };
23    } finally {
24        lineReader?.close();
25        stream?.destroy();
26    }
27};
