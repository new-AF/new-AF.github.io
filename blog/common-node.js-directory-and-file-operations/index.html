📁 List Directory Content

listDir.mjs
import { readdir } from "node:fs/promises";

export const listDir = async (path) => {
    try {
        const array = await readdir(path); // string[]
        return { success: true, array };
    } catch (error) {
        return {
            success: false,
            array: [],
            error,
        };
    }
};
listDir.demo.mjs
import { listDir } from "./listDir.mjs";

console.log("Valid Dir", await listDir("."));
console.log("Not Found Dir", await listDir("./not-found"));

📖 Read File Line By Line

readLines.mjs
import { createReadStream } from "node:fs";
import { createInterface } from "node:readline/promises";

export const readLines = async function* (filePath, encoding = "utf8") {
    let stream;
    let lineReader;

    try {
        stream = createReadStream(filePath, encoding);
        lineReader = createInterface({ input: stream, crlfDelay: Infinity });

        for await (const line of lineReader) {
            yield { success: true, line, error: null };
        }
    } catch (error) {
        yield {
            success: false,
            line: null,
            error,
        };
    } finally {
        lineReader?.close();
        stream?.destroy();
    }
};
readLines.demo.js
import { readLines } from "./readLines.mjs";

for await (const { success, line, error } of readLines("./readLines.mjs")) {
    console.log({ success, line, error });
}
for await (const { success, line, error } of readLines("./not-found")) {
    console.log({ success, line, error });
}
